package ch.zhaw.it.pm4.discordmanagerbe.bots.todo;

import ch.zhaw.it.pm4.discordmanagerbe.botmaker.JdaBotEntry;
import ch.zhaw.it.pm4.discordmanagerbe.botmaker.JdaBotService;
import ch.zhaw.it.pm4.discordmanagerbe.botmaker.annotation.SlashCommandBotType;
import ch.zhaw.it.pm4.discordmanagerbe.data.entities.TaskEntity;
import ch.zhaw.it.pm4.discordmanagerbe.data.entities.TaskReminderEntity;
import jakarta.transaction.Transactional;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

/**
 * This class is responsible for scheduling and checking reminders for tasks.
 * It checks for due reminders and tasks at a fixed rate and sends notifications to users.
 */
@Service
public class ReminderScheduler {

    private final TaskService taskService;

    private final JdaBotService jdaBotService;

    /**
     * Constructor for ReminderScheduler.
     * @param taskService    The task service instance for managing tasks and reminders.
     */
    @Autowired
    public ReminderScheduler(JdaBotService jdaBotService, TaskService taskService) {
        this.jdaBotService = jdaBotService;
        this.taskService = taskService;
    }

    /**
     * This method is scheduled to run every 30 seconds.
     * It checks for due reminders and tasks, and sends notifications to users.
     */
    @Scheduled(fixedRate = 30_000)
    @Transactional
    public void checkReminders() {
        long now = System.currentTimeMillis();

        List<TaskReminderEntity> reminders = taskService.getTaskReminderRepository().findAllDueReminder(now);
        List<TaskEntity> tasks = taskService.getTaskRepository().findAllDueTasks(now);

        for(TaskReminderEntity reminder : reminders) {
            taskService.deleteTaskReminderById(reminder.getId());
            triggerReminder(reminder.getTask(), "⏰ Reminder for your task: \n");
        }

        for(TaskEntity task : tasks) {
            taskService.deleteTaskById(task.getTaskId());
            triggerReminder(task, "❌ Your task has been deleted because the time is up: \n");
        }
    }

    /**
     * This method triggers a reminder message for the user for a specific task.
     * @param task    The task entity for which the reminder is being sent.
     * @param message The message to be sent to the user.
     */
    private void triggerReminder(TaskEntity task, String message) {
        String userId = task.getUserId();
        String title = task.getTitle();

        Optional<JdaBotEntry> botOpt = jdaBotService.getBot(SlashCommandBotType.TODO);
        if (botOpt.isEmpty()) {
            throw new IllegalStateException("ToDoTrackerJdaBot not found");
        }
        ToDoTrackerJdaBot bot = (ToDoTrackerJdaBot) botOpt.get().getBotInstance();

        bot.sendPrivateMessage(userId, title, message + task.getDescription());
    }
}
