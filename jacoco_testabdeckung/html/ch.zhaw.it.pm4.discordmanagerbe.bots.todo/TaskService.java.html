package ch.zhaw.it.pm4.discordmanagerbe.bots.todo;

import ch.zhaw.it.pm4.discordmanagerbe.data.entities.TaskEntity;
import ch.zhaw.it.pm4.discordmanagerbe.data.entities.TaskReminderEntity;
import ch.zhaw.it.pm4.discordmanagerbe.data.repositories.TaskReminderRepository;
import ch.zhaw.it.pm4.discordmanagerbe.data.repositories.TaskRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

/**
 * Service class for managing tasks and task reminders.
 */
@Service
public class TaskService {

    private final TaskRepository taskRepository;
    private final TaskReminderRepository taskReminderRepository;

    private Task task;

    /**
     * Constructor for TaskService.
     * @param taskRepository         Repository for task entities.
     * @param taskReminderRepository Repository for task reminder entities.
     */
    public TaskService(TaskRepository taskRepository, TaskReminderRepository taskReminderRepository) {
        this.taskRepository = taskRepository;
        this.taskReminderRepository = taskReminderRepository;
    }

    public TaskRepository getTaskRepository(){
        return taskRepository;
    }

    public TaskReminderRepository getTaskReminderRepository(){
        return taskReminderRepository;
    }

    public Task getTask() {
        return task;
    }

    public void setTask(Task task) {
        this.task = task;
    }

    /**
     * Saves a task entity to the database.
     * @param userId The ID of the user associated with the task.
     */
    public void saveTask(String userId) {
        TaskEntity taskEntity = new TaskEntity();
        taskEntity.setUserId(userId);
        taskEntity.setTitle(task.getTitle());
        taskEntity.setDescription(task.getDescription());
        taskEntity.setTimeToBeDone(task.getTimeToBeDone());
        taskRepository.save(taskEntity);
    }

    /**
     * Saves a task reminder entity to the database.
     * @param taskId The ID of the task associated with the reminder.
     * @param reminderTime The time for the reminder.
     */
    public void saveTaskReminder(Long taskId, long reminderTime){
        TaskReminderEntity taskReminder = new TaskReminderEntity();
        taskReminder.setTask(taskRepository.findById(taskId).orElseThrow(() -> new RuntimeException("Task not found")));
        taskReminder.setReminderTime(reminderTime);
        taskReminderRepository.save(taskReminder);
    }

    public TaskEntity getNewestTaskByUserId(String userId) {
        return taskRepository.findNewestTaskByUserId(userId);
    }

    public List<TaskEntity> getAllTasksByUserId(String userId) {
        return taskRepository.findAllTaskByUserId(userId);
    }

    /**
     * Retrieves a task entity by its ID.
     * @param taskId The ID of the task.
     * @return The task entity.
     */
    @Transactional
    public void deleteTaskById(long taskId) {
        TaskEntity task = taskRepository.findById(taskId)
                .orElseThrow(() -> new RuntimeException("Task not found"));
        taskRepository.delete(task);
    }

    /**
     * Retrieves a task reminder entity by its ID.
     * @param taskReminderId The ID of the task reminder.
     * @return The task reminder entity.
     */
    @Transactional
    public void deleteTaskReminderById(long taskReminderId) {
        taskReminderRepository.deleteTaskReminderById(taskReminderId);
    }
}
